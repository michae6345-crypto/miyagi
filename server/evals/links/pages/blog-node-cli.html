Build a notes CLI with Node.js, step by step

By Jun Watanabe · May 19, 2026

This is a long one: we build notes, a command-line notebook, from an empty folder to a tool with add, list, search and delete. Each section adds one command and runs it. Everything goes in notes.js.

Setup

Start a package so Node knows the project, with the default answers:

npm init -y

Storage

Notes live in a JSON file next to the script. Two helpers read and write it:

const fs = require("fs");
const FILE = "notes.json";

function load() {
  return fs.existsSync(FILE) ? JSON.parse(fs.readFileSync(FILE, "utf8")) : [];
}

function save(notes) {
  fs.writeFileSync(FILE, JSON.stringify(notes, null, 2));
}

Check that the file at least parses:

node --check notes.js

Adding notes

function add(text) {
  const notes = load();
  notes.push({ id: Date.now(), text });
  save(notes);
  console.log(`Saved note ${notes.length}`);
}

We'll wire up the command line at the end; for now, read on.

Listing notes

function list() {
  load().forEach((note, i) => console.log(`${i + 1}. ${note.text}`));
}
node notes.js list

Searching

function search(word) {
  const hits = load().filter((n) => n.text.toLowerCase().includes(word.toLowerCase()));
  hits.forEach((note) => console.log(`- ${note.text}`));
}
node notes.js search milk

Deleting

function remove(n) {
  const notes = load();
  notes.splice(n - 1, 1);
  save(notes);
}
node notes.js delete 1

The command line

const [cmd, ...rest] = process.argv.slice(2);
const commands = { add: () => add(rest.join(" ")), list, search: () => search(rest[0]), delete: () => remove(Number(rest[0])) };
(commands[cmd] || list)();
node notes.js add "buy milk"

Bonus: colours

npm install chalk@4
const chalk = require("chalk");
console.log(chalk.green("notes is ready"));
